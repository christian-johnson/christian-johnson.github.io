{{/* Image render hook — C2PA-aware. This is a thin override of Blowfish's default image render hook. Its only addition is the "RenderImageSigned" branch: when the referenced image appears in the data file written by the c2pa-hugo-pre-commit hook (data/c2pa-images.json), we emit a responsive that points directly at the pre-signed, pre-resized files in static/. Those are served byte-for-byte, so the C2PA signature on every size survives all the way to the browser. Any image NOT in that data file falls through to Blowfish's normal behavior (on-the-fly Resize + srcset), unchanged. Nothing else here is custom — keep it in sync with the theme's render-image.html if you upgrade Blowfish. */}} {{- define "RenderImageSigned" -}} {{/* Build a responsive srcset from the pre-signed static variants. The data file lists variants sorted ascending by width; the last is the full-size signed image. Every entry is a static passthrough, so signatures hold. */}} {{- $entry := .entry -}} {{- $alt := .alt -}} {{- $variants := $entry.variants -}} {{- $smallest := index $variants 0 -}} {{- $full := index $variants (sub (len $variants) 1) -}} {{- $srcset := slice -}} {{- range $variants -}} {{- $srcset = $srcset | append (printf "%s %dw" .src (int .width)) -}} {{- end -}} {{- end -}} {{- define "RenderImageSimple" -}} {{- $imgObj := .imgObj -}} {{- $src := .src -}} {{- $alt := .alt -}} {{- end -}} {{- define "RenderImageResponsive" -}} {{- $imgObj := .imgObj -}} {{- $alt := .alt -}} {{- $originalWidth := $imgObj.Width -}} {{- $img800 := $imgObj -}} {{- $img1280 := $imgObj -}} {{- if gt $originalWidth 800 -}} {{- $img800 = $imgObj.Resize "800x" -}} {{- end -}} {{- if gt $originalWidth 1280 -}} {{- $img1280 = $imgObj.Resize "1280x" -}} {{- end -}} {{- $srcset := printf "%s 800w, %s 1280w" $img800.RelPermalink $img1280.RelPermalink -}} {{- end -}} {{- define "RenderImageCaption" -}} {{- with .caption -}} {{- end -}} {{- end -}} {{- $disableImageOptimizationMD := .Page.Site.Params.disableImageOptimizationMD | default false -}} {{- $urlStr := .Destination | safeURL -}} {{- $url := urls.Parse $urlStr -}} {{- $altText := .Text -}} {{- $caption := .Title -}} {{- $isRemote := findRE "^(https?|data)" $url.Scheme -}} {{- $resource := "" -}} {{- /* Look up a pre-signed entry by the raw destination (e.g. /img/foo_signed.jpg). */ -}} {{- $signed := false -}} {{- $signedData := index site.Data "c2pa-images" -}} {{- if and (not $isRemote) $signedData -}} {{- $signed = index $signedData .Destination -}} {{- end -}} {{- if and (not $isRemote) (not $signed) -}} {{- $resource = or ($.Page.Resources.GetMatch $urlStr) (resources.Get $urlStr) -}} {{- end -}}
